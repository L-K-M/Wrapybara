package com.wrapybara.runtime;

import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * Colours for the strips under the system bars, from what the page reports: its
 * theme-color for the status bar, else its body or root background. The values
 * come from page JavaScript, so anything but a plain opaque colour is ignored
 * rather than guessed at.
 */
final class AndroidPageColors {
    /** WebView's own canvas colour, shown before a page paints. */
    static final int DEFAULT = 0xFFFFFFFF;

    private static final Pattern HEX = Pattern.compile("#([0-9a-fA-F]{3}|[0-9a-fA-F]{6})");
    private static final Pattern RGB = Pattern.compile(
            "rgba?\\((\\d{1,3}), ?(\\d{1,3}), ?(\\d{1,3})(?:, ?([0-9.]+))?\\)");
    // Relative luminance where black and white text have equal contrast (WCAG).
    private static final double EQUAL_CONTRAST_LUMINANCE = 0.179;

    final int statusBar;
    final int navigationBar;

    AndroidPageColors(int statusBar, int navigationBar) {
        this.statusBar = statusBar;
        this.navigationBar = navigationBar;
    }

    static AndroidPageColors choose(String themeColor, String bodyBackground, String rootBackground) {
        Integer background = firstOf(opaque(bodyBackground), opaque(rootBackground));
        Integer top = firstOf(opaque(themeColor), background);
        return new AndroidPageColors(top != null ? top : DEFAULT,
                background != null ? background : DEFAULT);
    }

    /** An opaque {@code #rgb}, {@code #rrggbb}, {@code rgb()} or {@code rgba()} colour as ARGB. */
    static Integer opaque(String value) {
        if (value == null) return null;

        String color = value.trim();
        Matcher hex = HEX.matcher(color);
        if (hex.matches()) {
            String digits = hex.group(1);
            if (digits.length() == 3) {
                digits = new String(new char[] {
                        digits.charAt(0), digits.charAt(0), digits.charAt(1), digits.charAt(1),
                        digits.charAt(2), digits.charAt(2)});
            }
            return 0xFF000000 | Integer.parseInt(digits, 16);
        }

        Matcher rgb = RGB.matcher(color);
        if (!rgb.matches()) return null;
        if (rgb.group(4) != null && !isOpaqueAlpha(rgb.group(4))) return null;

        int red = Integer.parseInt(rgb.group(1));
        int green = Integer.parseInt(rgb.group(2));
        int blue = Integer.parseInt(rgb.group(3));
        if (red > 255 || green > 255 || blue > 255) return null;

        return 0xFF000000 | red << 16 | green << 8 | blue;
    }

    /** Whether dark system bar icons read better than light ones on {@code color}. */
    static boolean isLight(int color) {
        double luminance = 0.2126 * linear(color >> 16 & 0xFF)
                + 0.7152 * linear(color >> 8 & 0xFF) + 0.0722 * linear(color & 0xFF);
        return luminance > EQUAL_CONTRAST_LUMINANCE;
    }

    private static boolean isOpaqueAlpha(String alpha) {
        try {
            return Double.parseDouble(alpha) >= 1;
        } catch (NumberFormatException error) {
            return false;
        }
    }

    private static double linear(int channel) {
        double value = channel / 255.0;
        return value <= 0.04045 ? value / 12.92 : Math.pow((value + 0.055) / 1.055, 2.4);
    }

    private static Integer firstOf(Integer preferred, Integer fallback) {
        return preferred != null ? preferred : fallback;
    }
}
